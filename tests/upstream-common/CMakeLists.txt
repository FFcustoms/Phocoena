cmake_minimum_required(VERSION 3.20)
project(DolphinPortableUnitTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DOLPHIN_SOURCE "${PORT_ROOT}/upstream/dolphin" CACHE PATH "Pinned Dolphin source")
add_subdirectory("${DOLPHIN_SOURCE}/Externals/gtest" googletest)
set(TEST_NAMES BitFieldTest BitSetTest BitUtilsTest FlagTest FixedSizeQueueTest SPSCQueueTest SwapTest)
foreach(name IN LISTS TEST_NAMES)
  list(APPEND TEST_FILES "${DOLPHIN_SOURCE}/Source/UnitTests/Common/${name}.cpp")
endforeach()
add_executable(upstream-common-tests ${TEST_FILES})
target_include_directories(upstream-common-tests PRIVATE
  "${DOLPHIN_SOURCE}/Source/Core" "${DOLPHIN_SOURCE}/Externals/fmt/fmt/include")
target_compile_definitions(upstream-common-tests PRIVATE FMT_HEADER_ONLY)
target_link_libraries(upstream-common-tests PRIVATE gtest_main)
enable_testing()
add_test(NAME upstream-common COMMAND upstream-common-tests)
